## [ERROR-4] 联合类型必须进行分支穷尽性检查

当 `switch` 处理联合类型（union type）时，必须在 `default` 分支中进行穷尽性检查，确保所有可能的值都被覆盖。当未来新增联合类型的成员时，程序会在编译期报错。

### 错误示例

```tsx
// 缺少 default 分支，新增 level 值时不会有任何提示，静默遗漏
type Level = 'A1' | 'A2' | 'B1';

function getDisplayName(level: Level): string | undefined {
  switch (level) {
    case 'A1':
      return 'A1';
    case 'A2':
      return 'A2';
    // 'B1' 被遗漏，没有任何编译期警告
  }
}
```

```tsx
// default 分支返回空值但未做穷尽性检查，新增值时同样静默遗漏
type Level = 'A1' | 'A2' | 'B1';

function getDisplayName(level: Level): string | undefined {
  switch (level) {
    case 'A1':
      return 'A1';
    case 'A2':
      return 'A2';
    default:
      return undefined; // 新增的值开发者无感知
  }
}
```

### 正确示例

```tsx
// 在 default 分支使用 never 类型进行穷尽性检查
type Level = 'A1' | 'A2' | 'B1';

function getDisplayName(level: Level): string | undefined {
  switch (level) {
    case 'A1':
      return 'A1';
    case 'A2':
      return 'A2';
    case 'B1':
      return 'B1';
    default:
      const _exhaustiveCheck: never = level;
      void _exhaustiveCheck;
      return undefined;
  }
}
```
